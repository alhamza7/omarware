### **الخطوة 5: نفذ هذا الكود في Console:**
```javascript
fetch('/api/items/search?q=برهان')
  .then(r => r.json())
  .then(d => {
    console.log('===== اختبار API =====');
    const item = d.items[0];
    console.log('الكود:', item?.item_code);
    console.log('الاسم:', item?.item_name);
    console.log('الباركود:', item?.barcode);
    console.log('وحدة القياس:', item?.uom);
    console.log('====================');
    
    if (item?.uom) {
      alert('✅ SUCCESS!\nوحدة القياس = ' + item.uom);
    } else {
      alert('❌ FAILED!\nوحدة القياس غير موجودة في API');
    }
  });
```
